### 1. Authentication Status Checking
Based on Django patterns from `quotewise/rest_api.py` and `views_htmx.py`:

```typescript
// src/auth/auth-checker.ts
interface AuthStatus {
    isAuthenticated: boolean;
    isStaff: boolean;           // Required for quote submission (user.is_staff)
    username?: string;
    sessionAge?: number;        // Time until session expires
}

interface AuthError {
    type: 'session_expired' | 'not_authenticated' | 'insufficient_privileges' | 'network_error';
    message: string;
    requiresLogin: boolean;
}

class AuthChecker {
    constructor(private apiClient: QuotewiseApiClient) {}
    
    async checkAuthStatus(): Promise<AuthStatus | AuthError> {
        try {
            // Create endpoint similar to existing pattern in rest_api.py
            const response = await this.apiClient.makeRequest<AuthStatusResponse>(
                '/api/v1/auth/status/'
            );
            
            return {
                isAuthenticated: response.is_authenticated,
                isStaff: response.is_staff,        // Matches Django user.is_staff
                username: response.username,
                sessionAge: response.session_age
            };
            
        } catch (error) {
            if (error instanceof AuthenticationError) {
                return {
                    type: 'not_authenticated',
                    message: 'Please log in to quotosaurus.com',
                    requiresLogin: true
                };
            }
            
            return {
                type: 'network_error', 
                message: 'Unable to check authentication status',
                requiresLogin: false
            };
        }
    }
    
    async waitForAuthChange(timeout = 30000): Promise<AuthStatus> {
        // Poll for authentication changes after login redirect
        const startTime = Date.now();
        
        while (Date.now() - startTime < timeout) {
            const status = await this.checkAuthStatus();
            
            if ('isAuthenticated' in status && status.isAuthenticated) {
                return status;
            }
            
            await new Promise(resolve => setTimeout(resolve, 1000));
        }
        
        throw new Error('Authentication timeout');
    }
}
```

### 2. Login Redirect Flow
Based on Django settings from `quotewise/settings/base.py` (lines 343, 437):

```typescript
// src/auth/login-handler.ts
interface LoginConfig {
    loginUrl: string;           // "/accounts/login/" from Django settings
    redirectUrl: string;        // "/" from LOGIN_REDIRECT_URL
    environment: 'dev' | 'staging' | 'prod';
}

class LoginHandler {
    private config: LoginConfig;
    
    constructor(environment: string) {
        this.config = this.getLoginConfig(environment);
    }
    
    private getLoginConfig(env: string): LoginConfig {
        const baseUrls = {
            development: 'http://localhost:8001',
            staging: 'https://staging.quotosaurus.com', 
            production: 'https://quotosaurus.com'
        };
        
        return {
            loginUrl: `${baseUrls[env]}/accounts/login/`,      // Django LOGIN_URL
            redirectUrl: `${baseUrls[env]}/`,                  // Django LOGIN_REDIRECT_URL
            environment: env as 'dev' | 'staging' | 'prod'
        };
    }
    
    async openLoginPage(): Promise<void> {
        // Open login page in new tab
        return new Promise((resolve, reject) => {
            chrome.tabs.create(
                { url: this.config.loginUrl, active: true },
                (tab) => {
                    if (chrome.runtime.lastError) {
                        reject(new Error(chrome.runtime.lastError.message));
                        return;
                    }
                    
                    // Listen for tab completion
                    this.watchForLoginCompletion(tab.id!, resolve, reject);
                }
            );
        });
    }
    
    private watchForLoginCompletion(
        tabId: number, 
        resolve: () => void, 
        reject: (error: Error) => void
    ): void {
        const onUpdated = (updatedTabId: number, changeInfo: chrome.tabs.TabChangeInfo) => {
            if (updatedTabId === tabId && changeInfo.status === 'complete') {
                chrome.tabs.get(tabId, (tab) => {
                    if (tab.url?.startsWith(this.config.redirectUrl)) {
                        // User successfully logged in and was redirected
                        chrome.tabs.onUpdated.removeListener(onUpdated);
                        chrome.tabs.onRemoved.removeListener(onRemoved);
                        resolve();
                    }
                });
            }
        };
        
        const onRemoved = (removedTabId: number) => {
            if (removedTabId === tabId) {
                chrome.tabs.onUpdated.removeListener(onUpdated);
                chrome.tabs.onRemoved.removeListener(onRemoved);
                reject(new Error('Login tab closed'));
            }
        };
        
        chrome.tabs.onUpdated.addListener(onUpdated);
        chrome.tabs.onRemoved.addListener(onRemoved);
        
        // Timeout after 5 minutes
        setTimeout(() => {
            chrome.tabs.onUpdated.removeListener(onUpdated);
            chrome.tabs.onRemoved.removeListener(onRemoved);
            reject(new Error('Login timeout'));
        }, 300000);
    }
}
```

### 3. Popup Integration
Handle authentication status in extension popup:

```typescript
// src/popup/popup.ts
class PopupController {
    private authChecker: AuthChecker;
    private loginHandler: LoginHandler;
    
    constructor() {
        this.authChecker = new AuthChecker(apiClient);
        this.loginHandler = new LoginHandler(ENVIRONMENT);
    }
    
    async initialize(): Promise<void> {
        const authStatus = await this.authChecker.checkAuthStatus();
        
        if ('type' in authStatus) {
            // Authentication error
            this.showLoginRequired(authStatus);
            return;
        }
        
        if (!authStatus.isAuthenticated) {
            this.showLoginRequired({
                type: 'not_authenticated',
                message: 'Please log in to quotosaurus.com',
                requiresLogin: true
            });
            return;
        }
        
        if (!authStatus.isStaff) {
            this.showInsufficientPrivileges();
            return;
        }
        
        // User is authenticated and has admin privileges
        this.showMainInterface(authStatus);
    }
    
    private showLoginRequired(error: AuthError): void {
        const container = document.getElementById('popup-container')!;
        container.innerHTML = `
            <div class="auth-required">
                <h3>Authentication Required</h3>
                <p>${error.message}</p>
                <button id="login-button" class="btn-primary">
                    Login to Quotosaurus
                </button>
            </div>
        `;
        
        document.getElementById('login-button')!.addEventListener('click', 
            () => this.handleLoginClick()
        );
    }
    
    private showInsufficientPrivileges(): void {
        const container = document.getElementById('popup-container')!;
        container.innerHTML = `
            <div class="insufficient-privileges">
                <h3>Admin Access Required</h3>
                <p>You need administrator privileges to submit quotes.</p>
                <p>Contact an administrator to request access.</p>
            </div>
        `;
    }
    
    private async handleLoginClick(): Promise<void> {
        try {
            const loginButton = document.getElementById('login-button')! as HTMLButtonElement;
            loginButton.disabled = true;
            loginButton.textContent = 'Opening login page...';
            
            await this.loginHandler.openLoginPage();
            
            loginButton.textContent = 'Checking authentication...';
            
            // Wait for authentication to complete
            const authStatus = await this.authChecker.waitForAuthChange();
            
            if (authStatus.isStaff) {
                this.showMainInterface(authStatus);
            } else {
                this.showInsufficientPrivileges();
            }
            
        } catch (error) {
            const loginButton = document.getElementById('login-button')! as HTMLButtonElement;
            loginButton.disabled = false;
            loginButton.textContent = 'Login to Quotosaurus';
            
            this.showError(`Login failed: ${error.message}`);
        }
    }
}
```

### 4. Background Script Integration
Handle authentication state changes:

```typescript
// src/background/service-worker.ts
class AuthenticationManager {
    private authChecker: AuthChecker;
    private lastAuthStatus: AuthStatus | null = null;
    
    constructor() {
        this.authChecker = new AuthChecker(apiClient);
        this.startPeriodicCheck();
    }
    
    private startPeriodicCheck(): void {
        // Check auth status every 5 minutes
        setInterval(async () => {
            try {
                const currentStatus = await this.authChecker.checkAuthStatus();
                
                if ('isAuthenticated' in currentStatus) {
                    if (this.hasAuthStatusChanged(currentStatus)) {
                        this.handleAuthStatusChange(currentStatus);
                        this.lastAuthStatus = currentStatus;
                    }
                }
            } catch (error) {
                console.error('Background auth check failed:', error);
            }
        }, 300000); // 5 minutes
    }
    
    private hasAuthStatusChanged(newStatus: AuthStatus): boolean {
        if (!this.lastAuthStatus) return true;
        
        return (
            this.lastAuthStatus.isAuthenticated !== newStatus.isAuthenticated ||
            this.lastAuthStatus.isStaff !== newStatus.isStaff
        );
    }
    
    private handleAuthStatusChange(newStatus: AuthStatus): void {
        // Notify popup if open
        chrome.runtime.sendMessage({
            type: 'AUTH_STATUS_CHANGED',
            status: newStatus
        });
        
        // Update extension badge
        if (newStatus.isAuthenticated && newStatus.isStaff) {
            chrome.action.setBadgeText({ text: '' });
            chrome.action.setTitle({ title: 'Quotewise Extension - Ready' });
        } else {
            chrome.action.setBadgeText({ text: '!' });
            chrome.action.setTitle({ title: 'Quotewise Extension - Login Required' });
        }
    }
}
```

### 5. Required API Endpoint
Need to create this endpoint in quotewise/rest_api.py:

```python
# Add to quotewise/rest_api.py
@api_view(["GET"])
@permission_classes([IsAuthenticated])
def api_auth_status(request):
    """
    API endpoint for checking authentication status and user privileges.
    
    Used by Chrome extension to verify session authentication and admin access.
    
    Returns:
        Response: JSON response with authentication details:
            {
                "is_authenticated": true,
                "is_staff": true,
                "username": "admin",
                "session_age": 1814400
            }
    """
    user = request.user
    session = request.session
    
    # Calculate session age (time until expiration)
    session_age = getattr(settings, 'SESSION_COOKIE_AGE', 1814400)
    if hasattr(session, 'get_expiry_age'):
        session_age = session.get_expiry_age()
    
    return Response({
        "is_authenticated": True,  # Always true if @permission_classes([IsAuthenticated])
        "is_staff": user.is_staff,
        "username": user.username,
        "session_age": session_age
    })
```
