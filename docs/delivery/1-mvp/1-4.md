```typescript
function getCookie(name: string): string | null {
    if (!document.cookie) return null;
    const cookie = document.cookie
        .split(';') // Split cookies
        .map(c => c.trim()) // Trim whitespace
        .find(c => c.startsWith(name + '=')); // Find the correct cookie
    return cookie ? decodeURIComponent(cookie.split('=').slice(1).join('=')) : null;
}

async function getCSRFToken(): Promise<string | null> {
    // Get from cookie first (line 729 pattern)
    let token = getCookie('csrftoken');
    
    if (!token) {
        // Fetch from Django if cookie unavailable
        try {
            const response = await fetch(`${API_BASE_URL}/csrf/`, {
                credentials: 'include'
            });
            const data = await response.json();
            token = data.csrfToken;
        } catch (error) {
            console.error('Failed to fetch CSRF token:', error);
            return null;
        }
    }
    
    return token;
}
```

### 3. API Client Implementation
Following patterns from `quotewise/rest_api.py` and `quote_collection.js`:

```typescript
interface QuotewiseApiClient {
    baseUrl: string;
    searchOriginators(query: string, limit?: number): Promise<OriginatorSearchResult[]>;
    checkQuoteDuplicate(text: string, originatorId?: number): Promise<DuplicateCheckResult>;
    submitQuote(quoteData: QuoteSubmissionData): Promise<QuoteSubmissionResult>;
    checkAuthStatus(): Promise<AuthStatusResult>;
}

class QuotewiseApiClientImpl implements QuotewiseApiClient {
    constructor(private baseUrl: string) {}
    
    private async makeRequest<T>(
        endpoint: string, 
        options: RequestInit = {}
    ): Promise<T> {
        const csrfToken = await getCSRFToken();
        
        const defaultHeaders = {
            'Content-Type': 'application/json',
            'X-CSRFToken': csrfToken || '',  // Line 736 pattern
            'X-Requested-With': 'XMLHttpRequest'
        };
        
        const requestOptions: RequestInit = {
            credentials: 'include',  // Include session cookies
            headers: { ...defaultHeaders, ...(options.headers || {}) },
            ...options
        };
        
        const response = await fetch(`${this.baseUrl}${endpoint}`, requestOptions);
        
        if (!response.ok) {
            if (response.status === 401 || response.status === 403) {
                throw new AuthenticationError('Authentication required');
            }
            throw new ApiError(`API error: ${response.status}`);
        }
        
        return response.json();
    }
    
    async searchOriginators(query: string, limit = 10): Promise<OriginatorSearchResult[]> {
        // Uses existing endpoint from quotewise/rest_api.py:894-975
        const params = new URLSearchParams({ q: query, limit: limit.toString() });
        const result = await this.makeRequest<{results: OriginatorSearchResult[]}>(
            `/api/v1/originators/search/?${params}`
        );
        return result.results;
    }

    async checkAuthStatus(): Promise<AuthStatusResult> {
        try {
            const result = await this.makeRequest<AuthStatusResult>('/api/v1/auth/status/');
            return result;
        } catch (error) {
            if (error instanceof AuthenticationError) {
                return { isAuthenticated: false };
            }
            throw error;
        }
    }

    async checkQuoteDuplicate(text: string, originatorId?: string): Promise<DuplicateCheckResult> {
        const payload = { 
            text, 
            originator_id: originatorId ? parseInt(originatorId) : undefined 
        };
        return this.makeRequest<DuplicateCheckResult>('/api/v1/quotes/check-duplicate/', {
            method: 'POST',
            body: JSON.stringify(payload)
        });
    }

    async submitQuote(quoteData: QuoteSubmissionRequest): Promise<QuoteSubmissionResult> {
        try {
            const result = await this.makeRequest<QuoteSubmissionResult>('/api/v1/quotes/', {
                method: 'POST',
                body: JSON.stringify(quoteData)
            });
            return { success: true, message: 'Quote submitted successfully', quoteId: result.id };
        } catch (error) {
            return { 
                success: false, 
                message: error.message || 'Failed to submit quote',
                error: error.message 
            };
        }
    }
}
```

### 4. Service Worker Integration
The API client must integrate with the existing service worker from Task 1-1 to handle popup messages:

```typescript
// src/background/api-handler.ts
import { QuotewiseApiClientImpl } from '../api/quotewise-api';

class ApiHandler {
    private apiClient: QuotewiseApiClientImpl;
    
    constructor() {
        const environment = this.detectEnvironment();
        const config = ENVIRONMENTS[environment];
        this.apiClient = new QuotewiseApiClientImpl(config.apiBaseUrl);
        
        // Listen for popup messages
        chrome.runtime.onMessage.addListener(this.handleMessage.bind(this));
    }
    
    private async handleMessage(
        message: ExtensionMessage, 
        sender: chrome.runtime.MessageSender, 
        sendResponse: (response: any) => void
    ): Promise<void> {
        try {
            switch (message.type) {
                case MessageType.CHECK_AUTH_STATUS:
                    const authStatus = await this.apiClient.checkAuthStatus();
                    sendResponse(authStatus);
                    break;
                    
                case MessageType.SEARCH_ORIGINATORS:
                    const results = await this.apiClient.searchOriginators(
                        message.data.query, 
                        message.data.limit
                    );
                    sendResponse({ success: true, results });
                    break;
                    
                case MessageType.CHECK_DUPLICATE_QUOTE:
                    const duplicateResult = await this.apiClient.checkQuoteDuplicate(
                        message.data.text,
                        message.data.originatorId
                    );
                    sendResponse({ success: true, ...duplicateResult });
                    break;
                    
                case MessageType.SUBMIT_QUOTE:
                    const submissionResult = await this.apiClient.submitQuote(message.data);
                    sendResponse(submissionResult);
                    break;
                    
                default:
                    sendResponse({ error: 'Unknown message type' });
            }
        } catch (error) {
            console.error('API error:', error);
            sendResponse({ 
                success: false, 
                error: error.message || 'API request failed' 
            });
        }
    }
    
    private detectEnvironment(): string {
        // Detect based on extension context or manifest
        // Default to development for testing
        return process.env.NODE_ENV === 'production' ? 'production' : 'development';
    }
}

// Initialize in service worker
new ApiHandler();
```
