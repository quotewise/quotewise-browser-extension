### 2. Add Sighting API Method

**Modify `src/api/quotewise-api.ts`**:
```typescript
export interface SightingSubmission {
  source_url: string;
  platform_code: string;
  engagement_metrics?: {
    likes?: number;
    retweets?: number;
    replies?: number;
  };
}

async addQuoteSighting(
  quoteId: string,
  sighting: SightingSubmission
): Promise<{ success: boolean; sighting_id?: string }> {
  return this.makeRequest(`/api/v1/quotes/${quoteId}/sightings/`, {
    method: 'POST',
    body: JSON.stringify(sighting)
  });
}
```

```typescript
// Add handler for collect-quote action
private setupCollectQuoteHandler(): void {
  // Listen for collect-quote button clicks
  this.duplicateDisplay.onCollectQuote = async (quoteId: string) => {
    try {
      const sighting: SightingSubmission = {
        source_url: this.state.tweetData.url,
        platform_code: 'TX',
        engagement_metrics: {
          likes: this.state.tweetData.platform_data.likes_count,
          retweets: this.state.tweetData.platform_data.retweets_count
        }
      };

      await apiClient.addQuoteSighting(quoteId, sighting);

      this.updateCollectionBadge({ state: 'sighting_added' });
      this.showSuccess('Sighting added to existing quote!');
    } catch (error) {
      this.updateCollectionBadge({ state: 'error' });
      this.showError('Failed to add sighting');
    }
  };
}
```
